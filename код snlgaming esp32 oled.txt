// ========== Theme Drawing Functions ==========
inline void setDrawColor() {
  if (whiteTheme) display.setTextColor(BLACK);
  else display.setTextColor(WHITE);
}

inline void setInverseColor() {
  if (whiteTheme) display.setTextColor(WHITE);
  else display.setTextColor(BLACK);
}

inline void drawRect(int x, int y, int w, int h, bool fill) {
  if (fill) {
    if (whiteTheme) display.fillRect(x, y, w, h, BLACK);
    else display.fillRect(x, y, w, h, WHITE);
  } else {
    if (whiteTheme) display.drawRect(x, y, w, h, BLACK);
    else display.drawRect(x, y, w, h, WHITE);
  }
}

inline void drawCircle(int x, int y, int r, bool fill) {
  if (fill) {
    if (whiteTheme) display.fillCircle(x, y, r, BLACK);
    else display.fillCircle(x, y, r, WHITE);
  } else {
    if (whiteTheme) display.drawCircle(x, y, r, BLACK);
    else display.drawCircle(x, y, r, WHITE);
  }
}

inline void drawTriangle(int x1, int y1, int x2, int y2, int x3, int y3, bool fill) {
  if (fill) {
    if (whiteTheme) display.fillTriangle(x1, y1, x2, y2, x3, y3, BLACK);
    else display.fillTriangle(x1, y1, x2, y2, x3, y3, WHITE);
  } else {
    if (whiteTheme) display.drawTriangle(x1, y1, x2, y2, x3, y3, BLACK);
    else display.drawTriangle(x1, y1, x2, y2, x3, y3, WHITE);
  }
}

inline void drawLine(int x1, int y1, int x2, int y2) {
  if (whiteTheme) display.drawLine(x1, y1, x2, y2, BLACK);
  else display.drawLine(x1, y1, x2, y2, WHITE);
}

// ========== EEPROM Functions ==========
void initEEPROM() {
  EEPROM.begin(64);
  int signature = EEPROM.read(EE_SIGNATURE);
  if (signature != 0xAA) {
    EEPROM.write(EE_SNAKE, 0);
    EEPROM.write(EE_PONG, 0);
    EEPROM.write(EE_CAR, 0);
    EEPROM.write(EE_DINO, 0);
    EEPROM.write(EE_SHOOTER, 0);
    EEPROM.write(EE_FLAPPY, 0);
    EEPROM.write(EE_ARKANOID, 0);
    EEPROM.write(EE_TANKS, 0);
    EEPROM.write(EE_TETRIS_HIGH, 0);  
    EEPROM.write(EE_THEME, 0);
    EEPROM.write(EE_RANDOM_MIN, 1);
    EEPROM.write(EE_RANDOM_MAX, 100);
    EEPROM.write(EE_PET_SCORE, 0);
    EEPROM.write(EE_PET_HUNGER, 50);
    EEPROM.write(EE_PET_MOOD, 50);
    EEPROM.write(EE_PET_LEVEL, 1);    
    EEPROM.write(EE_BOOT_MUSIC, 1);   
    EEPROM.write(EE_BOOT_SPEED, 3);   
    EEPROM.write(EE_SIGNATURE, 0xAA);
    EEPROM.commit();
  }
  eepromInitialized = true;
}



void loadHighScores() {
  initEEPROM();
  
  highSnake = EEPROM.read(EE_SNAKE);
  highPong = EEPROM.read(EE_PONG);
  highCar = EEPROM.read(EE_CAR);
  highDino = EEPROM.read(EE_DINO);
  highShooter = EEPROM.read(EE_SHOOTER);
  highFlappy = EEPROM.read(EE_FLAPPY);
  highArkanoid = EEPROM.read(EE_ARKANOID);
  highTanks = EEPROM.read(EE_TANKS);
  tetrisHighScore = EEPROM.read(EE_TETRIS_HIGH);
  
  // Валидация тетриса
  if (tetrisHighScore > 100) tetrisHighScore = 0;
  
  whiteTheme = EEPROM.read(EE_THEME);
  randomMin = EEPROM.read(EE_RANDOM_MIN);
  randomMax = EEPROM.read(EE_RANDOM_MAX);
  
  // ========== ЗАГРУЗКА ДАННЫХ ПИТОМЦА ==========
  petHunger = EEPROM.read(EE_PET_HUNGER);
  petMood = EEPROM.read(EE_PET_MOOD);
  petScore = EEPROM.read(EE_PET_SCORE);
  petLevel = EEPROM.read(EE_PET_LEVEL);
  
  // ========== ВАЛИДАЦИЯ ==========
  if (petHunger < 0 || petHunger > 100) petHunger = 50;
  if (petMood < 0 || petMood > 100) petMood = 50;
  if (petScore < 0) petScore = 0;
  if (petScore > 999) petScore = 999;
  if (petLevel < 1 || petLevel > 3) petLevel = 1;
  
  // ========== ОБНОВЛЯЕМ УРОВЕНЬ ПО ОЧКАМ ==========
  updatePetLevel();  // <--- ДОБАВИТЬ ЭТУ СТРОКУ
  
  // Валидация остальных рекордов
  if (highSnake > 255) highSnake = 0;
  if (highPong > 255) highPong = 0;
  if (highCar > 255) highCar = 0;
  if (highDino > 255) highDino = 0;
  if (highShooter > 255) highShooter = 0;
  if (highFlappy > 255) highFlappy = 0;
  if (highArkanoid > 255) highArkanoid = 0;
  if (highTanks > 255) highTanks = 0;
  
  if (whiteTheme > 1) whiteTheme = false;
  if (randomMin < 1) randomMin = 1;
  if (randomMax < randomMin) randomMax = randomMin + 1;
  if (randomMax > 999) randomMax = 999;
}

void saveHighScore(int gameAddr, int score, int &highScore) {
  if (score > highScore && score <= 255) {
    highScore = score;
    EEPROM.write(gameAddr, highScore);
    EEPROM.commit();
    delay(10);
  }
}

void saveTheme() {
  EEPROM.write(EE_THEME, whiteTheme ? 1 : 0);
  EEPROM.commit();
}

void saveRandomRange() {
  EEPROM.write(EE_RANDOM_MIN, randomMin);
  EEPROM.write(EE_RANDOM_MAX, randomMax);
  EEPROM.commit();
}

void savePetData() {
  EEPROM.write(EE_PET_HUNGER, petHunger);
  EEPROM.write(EE_PET_MOOD, petMood);
  EEPROM.write(EE_PET_SCORE, petScore);
  EEPROM.write(EE_PET_LEVEL, petLevel);  
  EEPROM.commit();
  delay(10); 
}

// ========== ПЛАВНАЯ АНИМАЦИЯ ПРЯМОУГОЛЬНИКА ==========
void updateSmoothCursor(int targetIndex, int startY, int stepY) {
  targetMenuIndex = targetIndex;
  float targetY = startY + targetIndex * stepY;
  
  unsigned long now = millis();
  if (now - lastAnimUpdate > 16) { // ~60 FPS
    lastAnimUpdate = now;
    // Плавное движение (easing)
    menuSmoothY = menuSmoothY + (targetY - menuSmoothY) * animSpeed;
    
    // Допуск для остановки анимации
    if (abs(menuSmoothY - targetY) < 0.5) {
      menuSmoothY = targetY;
      currentMenuIndex = targetMenuIndex;
    }
  }
}

void drawSmoothCursor(int x, int w, int h, int startY, int stepY) {
  int cursorY = (int)menuSmoothY;
  // Рисуем прямоугольник в плавной позиции
  drawRect(x, cursorY - 2, w, h, true);
}

// ========== PET FUNCTIONS ==========
void updatePetLevel() {
  int newLevel = 1;
  
  if (petScore >= 200) {
    newLevel = 3;   // MAX
  } else if (petScore >= 100) {
    newLevel = 2;
  } else {
    newLevel = 1;
  }
  
  if (newLevel != petLevel && newLevel <= 3) {
    petLevel = newLevel;
    savePetData();
  }
}


void updatePet() {
  if (millis() - petLastUpdate > 60000) {
    petLastUpdate = millis();
    petHunger -= 2;
    if (petHunger < 0) petHunger = 0;
    
    // ========== ОБНОВЛЯЕМ УРОВЕНЬ ==========
    updatePetLevel(); 
    
    // ========== ОПРЕДЕЛЯЕМ ЭМОЦИЮ ==========
    if (petHunger < 20) {
      petEmotion = 2;     // грустный
    } else if (petHunger > 80) {
      petEmotion = 1;     // счастливый
    } else {
      petEmotion = 0;     // нормальный
    }
    
    // ========== СОХРАНЯЕМ ==========
    savePetData();       
  }
}

void initFeedGame() {
  petX = 60;
  heartActive = false;
  heartX = -100;
  heartY = -100;
  lastHeartSpawn = millis();
  lastHeartMove = millis();
}


void drawFeedGame() {
  setDrawColor();
  
  // Выбираем битмап по эмоции (которая теперь от голода)
  const unsigned char* petBitmap = petNormal;
  if (petEmotion == 1) petBitmap = petHappy;
  else if (petEmotion == 2) petBitmap = petSad;
  else petBitmap = petNormal;
  
  int px = petX - 12;
  int py = 40;
  
  if (whiteTheme) {
    display.drawBitmap(px, py, petBitmap, 24, 24, BLACK);
  } else {
    display.drawBitmap(px, py, petBitmap, 24, 24, WHITE);
  }
  
  // Сердечко (как было)
  if (heartActive && heartY > 15) {
    drawCircle(heartX-3, heartY-2, 3, true);
    drawCircle(heartX+3, heartY-2, 3, true);
    drawTriangle(heartX-4, heartY-1, heartX+4, heartY-1, heartX, heartY+4, true);
  }

  // ========== СООБЩЕНИЕ О ПОЙМКЕ ==========
  if (showCatchMessage && millis() - catchMessageTime < 1000) {
    display.setTextSize(1);
    setInverseColor();  // или setDrawColor()
    display.setCursor(80, 3);
    display.print("CAUGHT");
    setDrawColor();
  } else {
    showCatchMessage = false;
  }
}


void updateFeedGame() {
  // Управление питомцем
  if (digitalRead(BTN_LEFT) == LOW) {
    petX -= 10;
    if (petX < 20) petX = 20;
    lastButtonPress = millis();
  }
  if (digitalRead(BTN_RIGHT) == LOW) {
    petX += 10;
    if (petX > 108) petX = 108;
    lastButtonPress = millis();
  }
  
  // Спавн сердечка
  if (!heartActive && millis() - lastHeartSpawn > 1500) {
    heartActive = true;
    heartX = random(25, 103);
    heartY = 15;
    lastHeartSpawn = millis();
  }
  
  // Движение сердечка
  if (heartActive && millis() - lastHeartMove > 50) {
    lastHeartMove = millis();
    heartY += 3;
    
    int petLeft = petX - 12;
    int petRight = petX + 12;
    int petTop = 40;
    int petBottom = 64;
    
    // Поймали сердечко
    if (heartY > petTop && heartY < petBottom && 
        heartX > petLeft && heartX < petRight) {
      heartActive = false;
      
      // ========== УВЕЛИЧИВАЕМ СЧЁТ И ГОЛОД ==========
      petScore++;
      petHunger = min(100, petHunger + 5);
      
      // ========== ОБНОВЛЯЕМ УРОВЕНЬ ==========
      updatePetLevel();  // <--- ДОБАВИТЬ ЭТУ СТРОКУ
      
      // ========== СОХРАНЯЕМ ==========
      savePetData();     // <--- ДОБАВИТЬ ЭТУ СТРОКУ
      
      // Показываем сообщение о поимке
      showCatchMessage = true;
      catchMessageTime = millis();
      
      heartX = -100;
      heartY = -100;
    }
    
    // Упало мимо
    if (heartY > 64) {
      heartActive = false;
      heartX = -100;
      heartY = -100;
    }
  }
}

void runFeedGame() {
  if (!feedGameActive) {
    initFeedGame();
    feedGameActive = true;
    return;
  }
  
  updateFeedGame();
  
  display.clearDisplay();
  if (whiteTheme) display.fillScreen(WHITE);
  setDrawColor();

  setInverseColor();
  display.setTextSize(1);
  display.setCursor(2, 3);

    feedGameActive = false;
    savePetData();
    changeState(PET_MENU);
    lastButtonPress = millis();
  }
}

// ========== GAME OVER FUNCTION ==========
void gameOver(String game, int score, int eeAddr, int &highScore) {
  gameActive = false;
  carGameActive = false;
  flappyActive = false;
  arkanoidActive = false;
  tankGameActive = false;
  sGameActive = false;
  tetrisActive = false;
  
  // ========== ДОБАВЛЯЕМ ОЧКИ ПИТОМЦУ ==========
  petScore += 10;
  petHunger += 5;
  if (petHunger > 100) petHunger = 100;
  if (petScore > 999) petScore = 999;
  
  // ========== ОБНОВЛЯЕМ УРОВЕНЬ ==========
  updatePetLevel();  // <--- ДОБАВИТЬ ЭТУ СТРОКУ
  
  // ========== СОХРАНЯЕМ ==========
  savePetData();
  
  // ========== СОХРАНЯЕМ РЕКОРД ==========
  if (game == "TETRIS") {
    if (score > tetrisHighScore) {
      tetrisHighScore = score;
      EEPROM.write(EE_TETRIS_HIGH, tetrisHighScore);
      EEPROM.commit();
    }
  } else {
    if (score > highScore && score <= 255) {
      highScore = score;
      EEPROM.write(eeAddr, highScore);
      EEPROM.commit();
    }
  }
  
  unsigned long startTime = millis();
  
  while (millis() - startTime < 3000) {
    display.clearDisplay();
    if (whiteTheme) display.fillScreen(WHITE);
    display.setTextSize(2);
    display.setCursor(10, 10);
    setDrawColor();
    display.println("GAME OVER");
    display.setTextSize(1);
    display.setCursor(20, 35);
    display.print("Game: "); display.print(game);
    display.setCursor(20, 45);
    display.print("Score: "); display.print(score);
    display.setCursor(20, 55);
    display.print("BACK: Skip");
    display.display();
    
    if (digitalRead(BTN_BACK) == LOW) {
      delay(200);
      changeState(GAMES_MENU);
      return;
    }
  }
  
  changeState(GAMES_MENU);
}

// ========== Boot Animation ==========
// ========== ПЕРЕМЕННЫЕ ДЛЯ НЕБЛОКИРУЮЩЕЙ МЕЛОДИИ ==========
bool melodyPlaying = false;
int melodyStep = 0;
unsigned long lastNoteTime = 0;

// DARK FANTASY (Mysteries of an Ancient Church)
// Мрачная, низкая, торжественная мелодия
int darkMelody[] = {
  147, 147, 165, 165, 175, 175, 196, 196,  // низкие органные ноты
  165, 165, 147, 147, 131, 131, 147, 147,  // нисходящий хорал
  196, 196, 220, 220, 247, 247, 262, 262,  // подъём к кульминации
  247, 247, 220, 220, 196, 196, 175, 175,  // спуск
  165, 165, 147, 147, 131, 131, 123, 123,  // финальные басы
  110, 110, 98, 98, 87, 87, 73, 73,        // глубокий финал
  73, 98, 110, 131, 147, 165, 196, 220,    // последний аккорд (восходящий)
  196, 175, 147, 131, 123, 110, 98, 73      // затухание до тишины
};

int darkDurations[] = {
  300, 300, 300, 300, 300, 300, 400, 600,
  300, 300, 300, 300, 300, 300, 400, 600,
  300, 300, 300, 300, 300, 300, 400, 600,
  300, 300, 300, 300, 300, 300, 300, 300,
  300, 300, 300, 300, 300, 300, 400, 600,
  400, 400, 400, 400, 400, 400, 500, 800,
  200, 200, 200, 200, 200, 200, 300, 400,
  300, 300, 300, 300, 300, 300, 400, 800
};

int melodyLength = 64;  // количество нот

void updateBootMelody() {
  if (!melodyPlaying) return;
  
  unsigned long now = millis();
  if (now - lastNoteTime >= darkDurations[melodyStep] + 30) {
    melodyStep++;
    if (melodyStep >= melodyLength) {
      melodyPlaying = false;
      noTone(SPEAKER_PIN);
      return;
    }
    tone(SPEAKER_PIN, darkMelody[melodyStep], darkDurations[melodyStep]);
    lastNoteTime = now;
  }
}

void runBootAnimation() {
  int bootDuration = 500 + (bootSpeed - 1) * 500;  // speed=1 → 500ms, speed=10 → 5000ms
  unsigned long elapsed = millis() - bootStartTime;
  
  static bool melodyStarted = false;
  if (bootMusic && !melodyStarted && !melodyPlaying) {
    melodyPlaying = true;
    melodyStep = 0;
    tone(SPEAKER_PIN, darkMelody[0], darkDurations[0]);
    lastNoteTime = millis();
    melodyStarted = true;
  }
  
  if (bootMusic) {
    updateBootMelody();
  }
  
  display.clearDisplay();
  if (whiteTheme) display.fillScreen(WHITE);
  setDrawColor();
  
  if (whiteTheme) {
    display.drawBitmap(0, 0, bootLogo, 128, 64, BLACK);
  } else {
    display.drawBitmap(0, 0, bootLogo, 128, 64, WHITE);
  }
  
  display.display();
  
  if (elapsed >= bootDuration) {
    changeState(MAIN_MENU);
  }
}

// ========== Loop ==========
void loop() {
  if (currentState == STARTUP) {
    runBootAnimation();
    return;
  }
  
  updatePet();

  // ========== ПАУЗА В ИГРАХ ==========
if (currentState == SNAKE || currentState == PONG || currentState == CAR || 
    currentState == DINO || currentState == SHOOTER || currentState == FLAPPY || 
    currentState == ARKANOID || currentState == TANKS || currentState == TETRIS) {
  
  if (digitalRead(BTN_BACK) == LOW && millis() - lastButtonPress > DEBOUNCE_DELAY) {
    if (gamePaused) {
      gamePaused = false;
      gameActive = false;
      changeState(GAMES_MENU);
      lastButtonPress = millis();
      return;
    } else {
      gamePaused = true;
      lastButtonPress = millis();
      return;
    }
  }
  
  if (gamePaused && digitalRead(BTN_SELECT) == LOW && millis() - lastButtonPress > DEBOUNCE_DELAY) {
    gamePaused = false;
    lastButtonPress = millis();
    return;
  }
  
  if (gamePaused) {
    // Рисуем паузу ПОВЕРХ игры (без очистки)
    drawPauseScreen();
    display.display();  // ← обновляем экран, чтобы показать паузу
    return;
  }
}

  if (digitalRead(BTN_BACK) == LOW && millis() - lastButtonPress > DEBOUNCE_DELAY) {
    // Специальная обработка для CMD и CMD_RESULT
    if (currentState == CMD_LINE) {
      changeState(APPS_MENU);
      lastButtonPress = millis();
      return;
    }
    if (currentState == CMD_RESULT) {
      changeState(CMD_LINE);
      lastButtonPress = millis();
      return;
    }
     if (currentState == SOUND_PAD) {        
      changeState(APPS_MENU);
      lastButtonPress = millis();
      return;
    }
    if (currentState == BOOT_SPEED_SETTINGS) {    
      changeState(SETTINGS_MENU);
      lastButtonPress = millis();
      return;
    }
    if (currentState == MP3_PLAYER) {    
      changeState(APPS_MENU);
      lastButtonPress = millis();
      return;
    }
    if (currentState == RENDER_3D) {    
      changeState(APPS_MENU);
      lastButtonPress = millis();
      return;
    }

}
  if (digitalRead(BTN_BACK) == LOW && millis() - lastButtonPress > DEBOUNCE_DELAY) {
    if (currentState != MAIN_MENU && currentState != STARTUP && currentState != PET_MENU && currentState != PET_FEED) {
      if (currentState == SNAKE || currentState == PONG || currentState == CAR || 
          currentState == DINO || currentState == SHOOTER || currentState == FLAPPY || 
          currentState == ARKANOID || currentState == TANKS ) {
        changeState(GAMES_MENU);
      }
      else if (currentState == THEME_SETTINGS || currentState == SHOW_RECORDS || currentState == CODE_VIEWER) {
        changeState(SETTINGS_MENU);
      }
      else if (currentState == STOPWATCH || currentState == FLASHLIGHT || 
               currentState == CALCULATOR || currentState == RANDOM_GEN) {
        changeState(APPS_MENU);
      }
      else if (currentState == ABOUT) {
        changeState(SETTINGS_MENU);
      }
      else {
        changeState(MAIN_MENU);
      }
      lastButtonPress = millis();
      return;
    }
  }
  
  if (codeScrolling && millis() - lastCodeScroll > 30) {
    lastCodeScroll = millis();
    codeLine++;
    if (codeLine >= 35 - 3) codeLine = 0;
  }
  
  switch (currentState) {
    case MAIN_MENU: runMainMenu(); break;
    case GAMES_MENU: runGamesMenu(); break;
    case SETTINGS_MENU: runSettingsMenu(); break;
    case APPS_MENU: runAppsMenu(); break;
    case SNAKE: runSnake(); break;
    case PONG: runPong(); break;
    case CAR: runCarGame(); break;
    case DINO: runDinoGame(); break;
    case SHOOTER: runShooter(); break;
    case FLAPPY: runFlappy(); break;
    case ARKANOID: runArkanoid(); break;
    case TANKS: runTanks(); break;
    case STOPWATCH: runStopwatch(); break;
    case FLASHLIGHT: runFlashlight(); break;
    case CALCULATOR: runCalculator(); break;
    case RANDOM_GEN: runRandomGen(); break;
    case CODE_VIEWER: runCodeViewer(); break;
    case PET_MENU: runPetMenu(); break;
    case PET_FEED: runFeedGame(); break;
    case ABOUT: runAbout(); break;
    case SHOW_RECORDS: showRecords(); break;
    case THEME_SETTINGS: runThemeSettings(); break;
    case CMD_LINE: runCmdLine(); break;
    case CMD_RESULT: runCmdResult(); break;
    case SOUND_PAD: runSoundPad(); break;
    case BOOT_SPEED_SETTINGS: runBootSpeedSettings(); break;
    case MP3_PLAYER: runMP3Player(); break;
    case RENDER_3D: runRender3D(); break;
    case TETRIS: runTetris(); break;
    default: break;
  }
}

// ========== CALCULATOR ==========
void evaluateExpression() {
  String expr = calcExpression;
  if (expr.length() == 0) {
    calcResult = "0";
    return;
  }
  
  // ========== ПРОВЕРКА НА НЕПРАВИЛЬНЫЙ ФОРМАТ ==========
  bool hasNumber = false;
  bool hasOperator = false;
  int operatorCount = 0;
  int lastChar = -1;
  
  for (int i = 0; i < expr.length(); i++) {
    char c = expr[i];
    if (c >= '0' && c <= '9') {
      hasNumber = true;
    } else if (c == '+' || c == '-' || c == '*' || c == '/') {
      operatorCount++;
      // Проверка: два оператора подряд
      if (i > 0 && (expr[i-1] == '+' || expr[i-1] == '-' || expr[i-1] == '*' || expr[i-1] == '/')) {
        calcResult = "ERROR";
        return;
      }
    } else {
      calcResult = "ERROR";
      return;
    }
  }
  
  // Проверка: нет чисел
  if (!hasNumber) {
    calcResult = "ERROR: no number";
    return;
  }
  
  // Проверка: начинается с оператора
  char first = expr[0];
  if (first == '+' || first == '-' || first == '*' || first == '/') {
    calcResult = "ERROR";
    return;
  }
  
  // Проверка: заканчивается оператором
  char last = expr[expr.length() - 1];
  if (last == '+' || last == '-' || last == '*' || last == '/') {
    calcResult = "ERROR";
    return;
  }
  
  // ========== ВЫЧИСЛЕНИЕ ==========
  float numbers[10];
  char operators[10];
  int numCount = 0, opCount = 0;
  String numStr = "";
  
  for (int i = 0; i < expr.length(); i++) {
    char c = expr[i];
    if ((c >= '0' && c <= '9') || c == '.') {
      numStr += c;
    } else {
      if (numStr.length() > 0) {
        numbers[numCount++] = numStr.toFloat();
        numStr = "";
      }
      operators[opCount++] = c;
    }
  }
  if (numStr.length() > 0) {
    numbers[numCount++] = numStr.toFloat();
  }
  
  float result = numbers[0];
  for (int i = 0; i < opCount; i++) {
    if (operators[i] == '+') result += numbers[i+1];
    else if (operators[i] == '-') result -= numbers[i+1];
    else if (operators[i] == '*') result *= numbers[i+1];
    else if (operators[i] == '/') {
      if (numbers[i+1] != 0) result /= numbers[i+1];
      else {
        calcResult = "ERROR";
        return;
      }
    }
  }
  
  if (result == (int)result) calcResult = String((int)result);
  else calcResult = String(result, 2);
}

  display.setCursor(10, 42);
  display.println("SEL:START/STOP");
  display.setCursor(10, 52);

// ========== SHOOTER GAME ==========
void initShooterGame() {
  sPlayerX = 52;           // центр экрана 64 - 12
  sPlayerHealth = 100;
  sScore = 0;
  sKillCount = 0;
  sGameActive = true;
  sEnemySpeed = 110;
  sEnemySpawnDelay = 2000;
  sEnemyCount = 0;
  
  for (int i = 0; i < S_MAX_BULLETS; i++) sBulletActive[i] = false;
  for (int i = 0; i < S_MAX_ENEMIES; i++) sEnemyActive[i] = false;
  
  sLastEnemySpawn = millis();
  sLastShot = millis();
  sLastEnemyMove = millis();
}

void spawnEnemy() {
  if (sEnemyCount >= S_MAX_ENEMIES) return;
  
  int slot = -1;
  for (int i = 0; i < S_MAX_ENEMIES; i++) {
    if (!sEnemyActive[i]) {
      slot = i;
      break;
    }
  }
  if (slot == -1) return;
  
  sEnemyX[slot] = random(10, 118);
  sEnemyY[slot] = -10;
  sEnemyHealth[slot] = 20;
  sEnemyActive[slot] = true;
  sEnemyCount++;
}

void shootShooter() {
  if (millis() - sLastShot < sShootDelay) return;
  
  int slot = -1;
  for (int i = 0; i < S_MAX_BULLETS; i++) {
    if (!sBulletActive[i]) {
      slot = i;
      break;
    }
  }
  if (slot == -1) return;
  
  // Пуля вылетает из центра битмапа игрока (24x24 → центр = 12)
  sBulletX[slot] = sPlayerX + 1;   // X центр
  sBulletY[slot] = 49;              // Y чуть выше битмапа
  sBulletActive[slot] = true;
  sLastShot = millis();
}

void updateShooterGame() {
  if (!sGameActive) return;
  
  if (millis() - sLastEnemyMove > sEnemySpeed) {
    sLastEnemyMove = millis();
    
    for (int i = 0; i < S_MAX_ENEMIES; i++) {
      if (!sEnemyActive[i]) continue;
      
      sEnemyY[i] += 3;
      
      if (sEnemyY[i] > 64) {
        sEnemyActive[i] = false;
        sEnemyCount--;
      }
      
      if (abs(sEnemyX[i] - sPlayerX) < 18 && abs(sEnemyY[i] - 52) < 18) {
  sPlayerHealth -= 20;
  sEnemyActive[i] = false;
  sEnemyCount--;
  if (sPlayerHealth <= 0) {
    sGameActive = false;
    gameOver("SHOOTER", sScore, EE_SHOOTER, highShooter);
    return;
        }
      }   
    }
  }
  
  for (int i = 0; i < S_MAX_BULLETS; i++) {
    if (!sBulletActive[i]) continue;
    
    sBulletY[i] -= 5;
    
    if (sBulletY[i] < 14) {
      sBulletActive[i] = false;
      continue;
    }
    
    for (int j = 0; j < S_MAX_ENEMIES; j++) {
      if (!sEnemyActive[j]) continue;
      
      if (abs(sBulletX[i] - sEnemyX[j]) < 10 && abs(sBulletY[i] - sEnemyY[j]) < 10) {
        sEnemyHealth[j] -= 10;
        sBulletActive[i] = false;
        
        if (sEnemyHealth[j] <= 0) {
          sEnemyActive[j] = false;
          sEnemyCount--;
          sScore += 10;
          sKillCount++;
          
          if (sKillCount % 5 == 0 && sEnemySpeed > 60) {
            sEnemySpeed -= 2;
          }
          if (sKillCount % 8 == 0 && sEnemySpawnDelay > 800) {
            sEnemySpawnDelay -= 100;
          }
        }
        break;
      }
    }
  }
  
  if (millis() - sLastEnemySpawn > sEnemySpawnDelay && sEnemyCount < S_MAX_ENEMIES) {
    sLastEnemySpawn = millis();
    spawnEnemy();
  }
}

  // Рисуем битмап игрока (24x24)
  int px = sPlayerX - 12;
  int py = 44;
  
  if (whiteTheme) {
    display.drawBitmap(px, py, playerBitmap, 24, 24, BLACK);
  } else {

  }
  
  // Пули
  for (int i = 0; i < S_MAX_BULLETS; i++) {
    if (sBulletActive[i]) {
      drawRect(sBulletX[i] - 1, sBulletY[i] - 1, 2, 4, true);
    }
  }
  
  // ========== ВРАГИ (БИТМАП ЗОМБИ 18x18) ==========
  for (int i = 0; i < S_MAX_ENEMIES; i++) {
    if (sEnemyActive[i]) {
      int ex = sEnemyX[i] - 9;   // центрируем 18px
      int ey = sEnemyY[i] - 9;   // центрируем 18px
      
      if (whiteTheme) {
        display.drawBitmap(ex, ey, enemyBitmap, 18, 18, BLACK);
      } else {
        display.drawBitmap(ex, ey, enemyBitmap, 18, 18, WHITE);
      }
    }
  }
}

void runShooter() {

  if (gamePaused) {
  drawPauseScreen();
  return;
}
  if (!sGameActive) {
    initShooterGame();
    return;
  }
  
  // Плавное движение
  if (digitalRead(BTN_LEFT) == LOW) {
    sPlayerX -= 7;
  }
  if (digitalRead(BTN_RIGHT) == LOW) {
    sPlayerX += 7;
  }
  
  // Границы (битмап 24px, левый край 0, правый 128-24=104)
  if (sPlayerX < 12) sPlayerX = 12;
  if (sPlayerX > 116) sPlayerX = 116;
  
  // Стрельба
  if (millis() - lastButtonPress > DEBOUNCE_DELAY) {
    if (digitalRead(BTN_SELECT) == LOW && millis() - sLastShot > sShootDelay) {
      shootShooter();
      lastButtonPress = millis();
    }
    
    if (digitalRead(BTN_BACK) == LOW) {
      sGameActive = false;
      gameOver("SHOOTER", sScore, EE_SHOOTER, highShooter);
      return;
    }
  }
  
if (millis() - sLastEnemyMove > sEnemySpeed) {
  sLastEnemyMove = millis();
  for (int i = 0; i < S_MAX_ENEMIES; i++) {
    if (sEnemyActive[i]) {
      sEnemyY[i] += 3;
      
      // Исчезают, если ушли за нижний край
      if (sEnemyY[i] > 64) {
        sEnemyActive[i] = false;
        sEnemyCount--;
      }
      
      // Проверка столкновения с игроком (Y игрока 44-68)
      if (sEnemyY[i] > 44 && sEnemyY[i] < 68 && abs(sEnemyX[i] - sPlayerX) < 12) {
        sPlayerHealth -= 20;
        sEnemyActive[i] = false;
        sEnemyCount--;
        if (sPlayerHealth <= 0) {
          sGameActive = false;
          gameOver("SHOOTER", sScore, EE_SHOOTER, highShooter);
          return;
        }
      }
    }
  }
}
  
  // Пули
  for (int i = 0; i < S_MAX_BULLETS; i++) {
    if (sBulletActive[i]) {
      sBulletY[i] -= 8;
      if (sBulletY[i] < 14) {
        sBulletActive[i] = false;
        continue;
      }
      for (int j = 0; j < S_MAX_ENEMIES; j++) {
        if (sEnemyActive[j] && abs(sBulletX[i] - sEnemyX[j]) < 10 && abs(sBulletY[i] - sEnemyY[j]) < 10) {
          sEnemyHealth[j] -= 10;
          sBulletActive[i] = false;
          if (sEnemyHealth[j] <= 0) {
            sEnemyActive[j] = false;
            sEnemyCount--;
            sScore += 1;
            sKillCount++;
            if (sKillCount % 5 == 0 && sEnemySpeed > 60) sEnemySpeed -= 2;
            if (sKillCount % 8 == 0 && sEnemySpawnDelay > 800) sEnemySpawnDelay -= 100;
          }
          break;
        }
      }
    }
  }
  
  // Спавн врагов
  if (millis() - sLastEnemySpawn > sEnemySpawnDelay && sEnemyCount < S_MAX_ENEMIES) {
    for (int i = 0; i < S_MAX_ENEMIES; i++) {
      if (!sEnemyActive[i]) {
        sEnemyX[i] = random(10, 118);
        sEnemyY[i] = 16;
        sEnemyHealth[i] = 20;
        sEnemyActive[i] = true;
        sEnemyCount++;
        sLastEnemySpawn = millis();
        break;
      }
    }
  }
  
  // Отрисовка
  display.clearDisplay();
  if (whiteTheme) display.fillScreen(WHITE);
  setDrawColor();
  
  drawRect(0, 0, 128, 10, true);
  setInverseColor();
  display.setTextSize(1);
  display.setCursor(2, 1);
  display.print("SCORE:"); display.print(sScore);
  display.setCursor(56, 1);
  display.print("HP:"); display.print(sPlayerHealth);
  display.setCursor(95, 1);
  display.print("K:"); display.print(sKillCount);
  setDrawColor();
  
  drawShooterGame();
  
  display.display();
}

  tankX = 60;
  tankY = 50;
  tankDirX = 0;
  tankDirY = -1;
  tankHealth = 10;
  tankScore = 0;
  tankGameActive = true;
  tankBulletActive = false;
  waveTransition = false;
  currentWave = 1;
  enemiesInWave = 3;
  enemiesRemaining = 3;
  
  for (int i = 0; i < 10; i++) {
    enemyHealth[i] = 0;
    enemyTanks[i][0] = 0;
    enemyTanks[i][1] = 0;
    enemyBullets[i][2] = 0;
  }
  
  for (int i = 0; i < enemiesInWave; i++) {
    enemyTanks[i][0] = 20 + (i % 3) * 30;
    enemyTanks[i][1] = 15 + (i / 3) * 20;
    enemyTanks[i][2] = random(-1, 2);
    enemyTanks[i][3] = random(-1, 2);
    while (enemyTanks[i][2] == 0 && enemyTanks[i][3] == 0) {
      enemyTanks[i][2] = random(-1, 2);
      enemyTanks[i][3] = random(-1, 2);
    }
    enemyHealth[i] = 1;
    enemyBullets[i][2] = 0;
  }
  
  lastTankMove = millis();
  lastTankShot = millis();

void showWaveNotification() {

  if (whiteTheme) display.fillScreen(WHITE);
  display.setTextSize(2);
  display.setCursor(30, 20);
  setDrawColor();
  display.print("WAVE");
  display.setCursor(80,20 );
  display.print(currentWave);
  display.setTextSize(1);
  display.setCursor(40, 40);
  display.print("GET READY!");
  display.display();
  delay(1500);
}

void spawnNextWave() {
  waveTransition = false;
  currentWave++;
  
  if (currentWave == 2) {
    enemiesInWave = 5;
  } else {
    enemiesInWave = 7;
  }
  
  enemiesRemaining = enemiesInWave;
  
  for (int i = 0; i < 10; i++) {
    enemyHealth[i] = 0;
    enemyTanks[i][0] = 0;
    enemyTanks[i][1] = 0;
    enemyBullets[i][2] = 0;
  }
  
  for (int i = 0; i < enemiesInWave; i++) {
    enemyTanks[i][0] = 15 + (i % 4) * 25;
    enemyTanks[i][1] = 15 + (i / 4) * 20;
    enemyTanks[i][2] = random(-1, 2);
    enemyTanks[i][3] = random(-1, 2);
    while (enemyTanks[i][2] == 0 && enemyTanks[i][3] == 0) {
      enemyTanks[i][2] = random(-1, 2);
      enemyTanks[i][3] = random(-1, 2);
    }
    enemyHealth[i] = 1;
    enemyBullets[i][2] = 0;
  }
  
  showWaveNotification();
}

void drawTank(int x, int y, int dirX, int dirY) {
  drawRect(x-5, y-5, 10, 10, true);
  drawRect(x-3, y-3, 6, 6, false);
  
  if (dirX == 0 && dirY == -1) {
    drawRect(x-1, y-7, 2, 3, true);
  } else if (dirX == 0 && dirY == 1) {
    drawRect(x-1, y+4, 2, 3, true);
  } else if (dirX == -1 && dirY == 0) {
    drawRect(x-7, y-1, 3, 2, true);
  } else if (dirX == 1 && dirY == 0) {
    drawRect(x+4, y-1, 3, 2, true);
  }
}

void runTanks() {

  if (gamePaused) {
  drawPauseScreen();
  return;
}
  if (!tankGameActive) return;
  
  if (!waveTransition && enemiesRemaining <= 0) {
    if (currentWave < 3) {
      waveTransition = true;
      spawnNextWave();
    } else {
      gameOver("TANKS", tankScore, EE_TANKS, highTanks);
      return;
    }
  }
  
  if (millis() - lastButtonPress > DEBOUNCE_DELAY) {
    if (digitalRead(BTN_SELECT) == LOW) {
      if (!tankBulletActive && millis() - lastTankShot > 400) {
        tankBulletX = tankX;
        tankBulletY = tankY;
        tankBulletDirX = tankDirX;
        tankBulletDirY = tankDirY;
        tankBulletActive = true;
        lastTankShot = millis();
      }
      lastButtonPress = millis();
    }
  }
  
  int newX = tankX;
  int newY = tankY;
  
  if (digitalRead(BTN_UP) == LOW) { newY -= 2; tankDirX = 0; tankDirY = -1; }
  if (digitalRead(BTN_DOWN) == LOW) { newY += 2; tankDirX = 0; tankDirY = 1; }
  if (digitalRead(BTN_LEFT) == LOW) { newX -= 2; tankDirX = -1; tankDirY = 0; }
  if (digitalRead(BTN_RIGHT) == LOW) { newX += 2; tankDirX = 1; tankDirY = 0; }
  
  if (newX > 15 && newX < 113 && newY > 15 && newY < 55) {
    tankX = newX;
    tankY = newY;
  }
  
  if (tankBulletActive) {
    tankBulletX += tankBulletDirX * 4;
    tankBulletY += tankBulletDirY * 4;
    if (tankBulletX < 0 || tankBulletX > 128 || tankBulletY < 12 || tankBulletY > 64) {
      tankBulletActive = false;
    }
    for (int i = 0; i < enemiesInWave; i++) {
      if (enemyHealth[i] > 0 && abs(tankBulletX - enemyTanks[i][0]) < 8 && abs(tankBulletY - enemyTanks[i][1]) < 8) {
        enemyHealth[i] = 0;
        tankBulletActive = false;
        tankScore++;
        enemiesRemaining--;
      }
    }
  }
  
  if (millis() - lastTankMove > tankSpeed) {
    lastTankMove = millis();
    for (int i = 0; i < enemiesInWave; i++) {
      if (enemyHealth[i] > 0) {
        int newEnemyX = enemyTanks[i][0] + enemyTanks[i][2] * 1;
        int newEnemyY = enemyTanks[i][1] + enemyTanks[i][3] * 1;
        if (newEnemyX > 15 && newEnemyX < 113 && newEnemyY > 15 && newEnemyY < 55) {
          enemyTanks[i][0] = newEnemyX;
          enemyTanks[i][1] = newEnemyY;
        } else {
          enemyTanks[i][2] = random(-1, 2);
          enemyTanks[i][3] = random(-1, 2);
          while (enemyTanks[i][2] == 0 && enemyTanks[i][3] == 0) {
            enemyTanks[i][2] = random(-1, 2);
            enemyTanks[i][3] = random(-1, 2);
          }
        }
        
        if (random(0, 100) < 2 && !enemyBullets[i][2]) {
          enemyBullets[i][0] = enemyTanks[i][0];
          enemyBullets[i][1] = enemyTanks[i][1];
          enemyBullets[i][2] = 1;
        }
      }
    }
  }
  
  for (int i = 0; i < enemiesInWave; i++) {
    if (enemyBullets[i][2]) {
      int dirX = (tankX - enemyBullets[i][0] > 0) ? 1 : -1;
      int dirY = (tankY - enemyBullets[i][1] > 0) ? 1 : -1;
      if (abs(tankX - enemyBullets[i][0]) > abs(tankY - enemyBullets[i][1])) {
        enemyBullets[i][0] += dirX * 3;
      } else {
        enemyBullets[i][1] += dirY * 3;
      }
      if (enemyBullets[i][0] < 0 || enemyBullets[i][0] > 128 || enemyBullets[i][1] < 12 || enemyBullets[i][1] > 64) {
        enemyBullets[i][2] = 0;
      }
      if (abs(enemyBullets[i][0] - tankX) < 8 && abs(enemyBullets[i][1] - tankY) < 8) {
        tankHealth--;
        enemyBullets[i][2] = 0;
        if (tankHealth <= 0) {
          gameOver("TANKS", tankScore, EE_TANKS, highTanks);
          return;
        }
      }
    }
  }
  
  display.clearDisplay();
  if (whiteTheme) display.fillScreen(WHITE);
  
  drawRect(0, 0, 128, 10, true);
  setInverseColor();
  display.setCursor(2, 1);
  display.print("SCORE:");
  display.print(tankScore);
  display.setCursor(50, 1);
  display.print("WAVE:");
  display.print(currentWave);
  display.setCursor(90, 1);
  display.print("HP:");
  display.print(tankHealth);
  setDrawColor();
  
  drawTank(tankX, tankY, tankDirX, tankDirY);
  
  for (int i = 0; i < enemiesInWave; i++) {
    if (enemyHealth[i] > 0) {
      drawTank(enemyTanks[i][0], enemyTanks[i][1], enemyTanks[i][2], enemyTanks[i][3]);
    }
  }
  
  if (tankBulletActive) {
    drawRect(tankBulletX-1, tankBulletY-1, 2, 2, true);
  }
  
  for (int i = 0; i < enemiesInWave; i++) {
    if (enemyBullets[i][2]) {
      drawRect(enemyBullets[i][0]-1, enemyBullets[i][1]-1, 2, 2, true);
    }
  }
  
  display.display();
}

  drawPauseScreen();
  return;
}

  if(!gameActive) return;

    if (digitalRead(BTN_SELECT) == LOW) { gameOver("SNAKE", snakeScore, EE_SNAKE, highSnake); return; }

    else if (digitalRead(BTN_DOWN) == LOW && dirY == 0) { dirX = 0; dirY = 1; lastButtonPress = millis(); }
    else if (digitalRead(BTN_LEFT) == LOW && dirX == 0) { dirX = -1; dirY = 0; lastButtonPress = millis(); }
    else if (digitalRead(BTN_RIGHT) == LOW && dirX == 0) { dirX = 1; dirY = 0; lastButtonPress = millis(); }
  }
  
  if(millis() - lastMoveTime > snakeSpeed){
    lastMoveTime = millis();
    for(int i = snakeLength-1; i > 0; i--){ snakeX[i] = snakeX[i-1]; snakeY[i] = snakeY[i-1]; }
    snakeX[0] += dirX*4; snakeY[0] += dirY*4;
    
    if(snakeX[0] < 0 || snakeX[0] >= 128 || snakeY[0] < 12 || snakeY[0] >= 64) { gameOver("SNAKE", snakeScore, EE_SNAKE, highSnake); return; }
    for(int i = 1; i < snakeLength; i++) if(snakeX[0] == snakeX[i] && snakeY[0] == snakeY[i]) { gameOver("SNAKE", snakeScore, EE_SNAKE, highSnake); return; }
    if(abs(snakeX[0]-foodX) < 4 && abs(snakeY[0]-foodY) < 4){
      if(snakeLength < MAX_SNAKE){ snakeLength++; snakeScore++; spawnFood(); if(snakeSpeed > 50) snakeSpeed -= 3; }
    }
  }
  
  display.clearDisplay();
  if (whiteTheme) display.fillScreen(WHITE);
  drawRect(0, 0, 128, 10, true);
  setInverseColor();
  display.setCursor(2,1); display.print("SCORE:"); display.print(snakeScore);
  setDrawColor();
  for(int i = 0; i < snakeLength; i++){ 
    if(i == 0) drawRect(snakeX[i], snakeY[i], 4, 4, true); 
    else drawRect(snakeX[i], snakeY[i], 3, 3, false); 
  }
  if(millis() % 400 < 200) drawCircle(foodX+2, foodY+2, 2, true);
  display.display();
}

// ========== CAR RACER ==========
void initCarGame() {
  playerLane = 1;
  carScore = 0;
  carSpeed = 90;          // начальная скорость выше
  carGameActive = true;
  
  for (int i = 0; i < 3; i++) {
    obstacleLane[i] = random(0, 3);
    obstacleY[i] = -20 - (i * 30);
  }
  
  lastCarMove = millis();
}


void drawCarGame() {
  int laneX[3] = {32, 64, 96};
  
  // Боковые линии дороги
  for (int i = 0; i < 3; i++) {
    drawLine(laneX[i] - 16, 12, laneX[i] - 16, 63);
  }
  drawLine(laneX[2] + 16, 12, laneX[2] + 16, 63);
  
  // ========== ВЕРТИКАЛЬНЫЙ ПУНКТИР ==========
  for (int i = 0; i < 3; i++) {
    for (int y = 16; y < 62; y += 8) {
      drawLine(laneX[i] - 1, y, laneX[i] - 1, y + 3);
    }
  }
  
  // ========== МАШИНА ИГРОКА (БИТМАП) ==========
  int px = laneX[playerLane] - 5;
  int py = 44;
  
  if (whiteTheme) {
    display.drawBitmap(px, py, carBitmap, 10, 16, BLACK);
  } else {
    display.drawBitmap(px, py, carBitmap, 10, 16, WHITE);
  }
  
  // ========== ЯМЫ ==========
  for (int i = 0; i < 3; i++) {
    if (obstacleY[i] > 12 && obstacleY[i] < 63) {
      int ox = laneX[obstacleLane[i]] - 5;
      int oy = obstacleY[i];
      
      drawRect(ox, oy, 10, 8, true);
      for (int y = oy + 2; y <= oy + 6; y += 2) {
        display.drawPixel(ox + 5, y, !whiteTheme);
      }
    }
  }
}
void runCarGame() {

  if (gamePaused) {
  drawPauseScreen();
  return;
}
  if (!carGameActive) {
    initCarGame();
    return;
  }
  
  // Управление
  if (millis() - lastButtonPress > DEBOUNCE_DELAY) {
    if (digitalRead(BTN_LEFT) == LOW && playerLane > 0) {
      playerLane--;
      lastButtonPress = millis();
    }
    if (digitalRead(BTN_RIGHT) == LOW && playerLane < 2) {
      playerLane++;
      lastButtonPress = millis();
    }
    if (digitalRead(BTN_SELECT) == LOW || digitalRead(BTN_BACK) == LOW) {
      carGameActive = false;
      gameOver("RACER", carScore, EE_CAR, highCar);
      return;
    }
  }
  
  // Движение ям
  if (millis() - lastCarMove > carSpeed) {
    lastCarMove = millis();
    
    for (int i = 0; i < 3; i++) {
      obstacleY[i] += 3;   // чуть быстрее
      
      // Новая яма
      if (obstacleY[i] > 63) {
        obstacleY[i] = 12;
        obstacleLane[i] = random(0, 3);
        carScore++;
        if (carSpeed > 60) carSpeed -= 1;
      }
      
      // Столкновение (игрок на Y = 44-60)
      if (obstacleY[i] > 42 && obstacleY[i] < 58 && obstacleLane[i] == playerLane) {
        carGameActive = false;
        gameOver("RACER", carScore, EE_CAR, highCar);
        return;
      }
    }
  }
  
  // Отрисовка
  display.clearDisplay();
  if (whiteTheme) display.fillScreen(WHITE);
  setDrawColor();
  
  // Верхняя панель
  drawRect(0, 0, 128, 10, true);
  setInverseColor();
  display.setCursor(2, 2);
  display.print("SCORE: ");
  display.print(carScore);
  setDrawColor();
  
  drawCarGame();
  
  display.display();
}
// ========== DINO GAME ==========
void initDinoGame() {
  dinoY=44; dinoVelocity=0; isJumping=false; obstacleX=128; dinoScore=0; dinoSpeed=50; dinoCrouch=false; gameActive=true; lastDinoUpdate=millis();
}

void runDinoGame() {

  if (gamePaused) {
  drawPauseScreen();
  return;
}
  if(!gameActive) return;
  
  if(millis() - lastButtonPress > DEBOUNCE_DELAY){
    if(digitalRead(BTN_UP) == LOW && !isJumping && !dinoCrouch){ isJumping=true; dinoVelocity=-8; lastButtonPress=millis(); }
    if(digitalRead(BTN_SELECT) == LOW){ gameOver("DINO", dinoScore, EE_DINO, highDino); return; }
  }
  
  if(digitalRead(BTN_DOWN) == LOW && !isJumping && dinoY >= 44) dinoCrouch = true;
  else dinoCrouch = false;
  
  if(millis() - lastDinoUpdate > dinoSpeed){
    lastDinoUpdate = millis();
    if(isJumping){ dinoY += dinoVelocity; dinoVelocity += 1; if(dinoY >= 44){ dinoY = 44; isJumping = false; dinoVelocity = 0; } }
    obstacleX -= 5;
    if(obstacleX < -10){ obstacleX = 128; dinoScore++; if(dinoSpeed > 25) dinoSpeed -= 1; }

// ========== FLAPPY BIRD ==========
void initFlappy() {
  flappyBirdY = 32;
  flappyBirdVelocity = 0;
  flappyPipeX = 128;

  flappyScore = 0;
  flappySpeed = 60;
  flappyActive = true;
  lastFlappyUpdate = millis();
}

void runFlappy() {
  if (gamePaused) {
  drawPauseScreen();
  return;
}
  if (!flappyActive) return;
  
  if (millis() - lastButtonPress > DEBOUNCE_DELAY) {
    if (digitalRead(BTN_SELECT) == LOW || digitalRead(BTN_UP) == LOW) {
      flappyBirdVelocity = -4;
      lastButtonPress = millis();

  if (millis() - lastFlappyUpdate > flappySpeed) {
    lastFlappyUpdate = millis();

    flappyBirdY += flappyBirdVelocity;

      if (flappyBirdY < flappyPipeTopHeight || flappyBirdY > flappyPipeTopHeight + flappyPipeGap) {
        gameOver("FLAPPY", flappyScore, EE_FLAPPY, highFlappy);
        return;
      }
    }
  }

  if (whiteTheme) display.fillScreen(WHITE);
  
  drawRect(flappyPipeX, 0, 10, flappyPipeTopHeight, true);
  drawRect(flappyPipeX, flappyPipeTopHeight + flappyPipeGap, 10, 64 - (flappyPipeTopHeight + flappyPipeGap), true);
  
  drawCircle(20, flappyBirdY, 4, true);
  drawCircle(18, flappyBirdY-2, 1, true);

// ========== ARKANOID ==========
void initArkanoid() {
  arkanoidPaddleX = 54;
  arkanoidBallX = 64;
  arkanoidBallY = 50;
  arkanoidBallDX = 1;
  arkanoidBallDY = -1;
  arkanoidScore = 0;
  arkanoidLives = 3;
  arkanoidActive = true;
  lastArkanoidUpdate = millis();
  
  for (int i = 0; i < 5; i++) {
    for (int j = 0; j < 12; j++) {
      arkanoidBricks[i][j] = 1;
    }
  }
}

void runArkanoid() {

  if (gamePaused) {
  drawPauseScreen();
  return;
}

  if (!arkanoidActive) return;
  
  if (millis() - lastButtonPress > DEBOUNCE_DELAY) {
    if (digitalRead(BTN_LEFT) == LOW) {
      arkanoidPaddleX -= 12;
      if (arkanoidPaddleX < 0) arkanoidPaddleX = 0;
      lastButtonPress = millis();
    }
    if (digitalRead(BTN_RIGHT) == LOW) {
      arkanoidPaddleX += 12;
      if (arkanoidPaddleX > 116) arkanoidPaddleX = 116;
      lastButtonPress = millis();
    }
    if (digitalRead(BTN_SELECT) == LOW) {
      gameOver("ARKANOID", arkanoidScore, EE_ARKANOID, highArkanoid);
      return;
    }
  }
  
  if (millis() - lastArkanoidUpdate > arkanoidSpeed) {
    lastArkanoidUpdate = millis();
    
    arkanoidBallX += arkanoidBallDX;
    arkanoidBallY += arkanoidBallDY;
    
    if (arkanoidBallX <= 2 || arkanoidBallX >= 125) {
      arkanoidBallDX = -arkanoidBallDX;
    }
    if (arkanoidBallY <= 10) {
      arkanoidBallDY = -arkanoidBallDY;
    }
    
    if (arkanoidBallY >= 58 && arkanoidBallY <= 62 && 
        arkanoidBallX >= arkanoidPaddleX && arkanoidBallX <= arkanoidPaddleX + 12) {
      arkanoidBallDY = -arkanoidBallDY;
      int hitPos = arkanoidBallX - arkanoidPaddleX;
      if (hitPos < 4) arkanoidBallDX = -1;
      else if (hitPos > 8) arkanoidBallDX = 1;
      else arkanoidBallDX = (hitPos - 6) * 0.3;
      if (arkanoidBallDX == 0) arkanoidBallDX = 1;
    }
    
    if (arkanoidBallY > 64) {
      arkanoidLives--;
      if (arkanoidLives <= 0) {
        gameOver("ARKANOID", arkanoidScore, EE_ARKANOID, highArkanoid);
        return;
      }
      arkanoidBallX = 64;
      arkanoidBallY = 50;
      arkanoidBallDX = 1;
      arkanoidBallDY = -1;
      delay(500); 
    }
    
    for (int i = 0; i < 5; i++) {
      for (int j = 0; j < 12; j++) {
        if (arkanoidBricks[i][j]) {
          int brickX = j * 10 + 4;
          int brickY = i * 5 + 14;
          if (arkanoidBallX + 3 >= brickX && arkanoidBallX - 3 <= brickX + 8 &&
              arkanoidBallY + 3 >= brickY && arkanoidBallY - 3 <= brickY + 3) {
            arkanoidBricks[i][j] = 0;
            arkanoidBallDY = -arkanoidBallDY;
            arkanoidScore++;
          }
        }
      }
    }
    
    bool allBricksGone = true;
    for (int i = 0; i < 5; i++) {
      for (int j = 0; j < 12; j++) {
        if (arkanoidBricks[i][j]) allBricksGone = false;
      }
    }
    if (allBricksGone) {
      gameOver("ARKANOID", arkanoidScore, EE_ARKANOID, highArkanoid);
      return;
    }
  }
  
  display.clearDisplay();
  if (whiteTheme) display.fillScreen(WHITE);
  drawRect(0, 0, 128, 10, true);
  setInverseColor();
  display.setCursor(2, 1);
  display.print("SCORE:"); display.print(arkanoidScore);
  display.setCursor(80, 1);
  display.print("LIVES:"); display.print(arkanoidLives);
  setDrawColor();
  
  for (int i = 0; i < 5; i++) {
    for (int j = 0; j < 12; j++) {
      if (arkanoidBricks[i][j]) {
        drawRect(j * 10 + 4, i * 5 + 14, 8, 3, true);
      }
    }
  }
  
  drawRect(arkanoidPaddleX, 60, 12, 3, true);
  drawCircle(arkanoidBallX, arkanoidBallY, 3, true);
  
  display.display();
}

//================ CMD ====================
String executeCmd(String cmd) {
  cmd.trim();
  
  // ========== ТРЕБУЕМ ПРЕФИКС snldo ==========
  if (!cmd.startsWith("snldo ")) {
  if (cmd == "snldo") {
    return "Error: missing command. Usage: snldo <command>";
  }
  return "Error: missing 'snldo' prefix. Example: snldo help";
}
  
  cmd = cmd.substring(6); 
  
  // ========== ИНФОРМАЦИОННЫЕ КОМАНДЫ ==========
  if (cmd == "help") {
    return "Commands: help, info, theme, reboot, clear, games, apps, pet, records, code";
  }
  else if (cmd == "info") {
    String info = "SNLGAMING \n";
    info += "RAM: " + String(ESP.getFreeHeap()) + " bytes\n";
    info += "Flash: " + String(ESP.getFlashChipSize() / 1024) + " KB\n";
    info += "Theme: " + String(whiteTheme ? "WHITE" : "DARK");
    return info;
  }
  
  // ========== КОМАНДЫ-ДЕЙСТВИЯ (открывают меню) ==========
  else if (cmd == "pet") {
    changeState(PET_MENU);
    return "";
  }
  else if (cmd == "records") {
    changeState(SHOW_RECORDS);
    return "";
  }
  else if (cmd == "games") {
    changeState(GAMES_MENU);
    return "";
  }
  else if (cmd == "apps") {
    changeState(APPS_MENU);
    return "";
  }
  else if (cmd == "settings") {
    changeState(SETTINGS_MENU);
    return "";
  }
  else if (cmd == "theme") {
    whiteTheme = !whiteTheme;
    saveTheme();
    return "Theme changed to " + String(whiteTheme ? "WHITE" : "DARK");
  }
  else if (cmd == "clear") {
    display.clearDisplay();
    return "";
  }
  else if (cmd == "reboot") {
    ESP.restart();
    return "";
  }
  else if (cmd == "code") {
    changeState(CODE_VIEWER);
    return "";
  }
  
  return "Unknown command. Type 'snldo help'";
}

void runCmdResult() {
  if (millis() - lastButtonPress > DEBOUNCE_DELAY) {
    if (digitalRead(BTN_BACK) == LOW) {
      changeState(CMD_LINE);   // ← возврат в командную строку
      cmdOutput = "";          // очищаем вывод
      lastButtonPress = millis();
      return;
    }
  }
  
  display.clearDisplay();
  if (whiteTheme) display.fillScreen(WHITE);
  setDrawColor();
  
  drawRect(0, 0, 128, 10, true);
  setInverseColor();
  display.setCursor(2, 2);
  display.print("COMMAND RESULT");
  setDrawColor();
  drawLine(0, 10, 128, 10);
  
  int y = 14;
  int lineStart = 0;
  
  for (int i = 0; i < cmdOutput.length() && y < 60; i++) {
    if (cmdOutput[i] == '\n' || i == cmdOutput.length() - 1) {
      String line = cmdOutput.substring(lineStart, i == cmdOutput.length() - 1 ? i + 1 : i);
      display.setCursor(2, y);
      display.print(line);
      y += 9;
      lineStart = i + 1;
    }
  }
  
  setInverseColor();
  display.setCursor(2, 58);
  display.print("BACK: Return");
  
  display.display();
}

void runCmdLine() {
  static float cmdSmoothX = 0;
  static float cmdSmoothY = 0;
  static int cmdTargetX = 0;
  static int cmdTargetY = 0;
  static unsigned long lastCmdAnim = 0;
  static unsigned long lastCmdMove = 0;
  
  if (!cmdActive) {
    cmdActive = true;
    cmdInput = "";
    cmdCursorX = 0;
    cmdCursorY = 0;
    cmdTargetX = 0;
    cmdTargetY = 0;
    cmdSmoothX = 2;
    cmdSmoothY = 28;
  }
  
  // Управление курсором
  if (millis() - lastCmdMove > 120) {
    bool moved = false;
    if (digitalRead(BTN_UP) == LOW && cmdTargetY > 0) { cmdTargetY--; moved = true; }
    if (digitalRead(BTN_DOWN) == LOW && cmdTargetY < 4) { cmdTargetY++; moved = true; }
    if (digitalRead(BTN_LEFT) == LOW && cmdTargetX > 0) { cmdTargetX--; moved = true; }
    if (digitalRead(BTN_RIGHT) == LOW && cmdTargetX < 7) { cmdTargetX++; moved = true; }
    
    if (moved) {
      lastCmdMove = millis();
      lastButtonPress = millis();
      cmdCursorX = cmdTargetX;
      cmdCursorY = cmdTargetY;
    }
  }
  
  // Ввод символов
  if (millis() - lastButtonPress > DEBOUNCE_DELAY) {
    if (digitalRead(BTN_SELECT) == LOW) {
      char selected = cmdKeyboard[cmdCursorY][cmdCursorX];
      if (selected == '@') {
        if (cmdInput.length() > 0) {
          cmdOutput = executeCmd(cmdInput);
          if (cmdOutput.length() > 0) {
            changeState(CMD_RESULT);
          } else {
            cmdInput = "";
          }
        }
      } else if (selected == '<') {
        if (cmdInput.length() > 0) {
          cmdInput.remove(cmdInput.length() - 1);
        }
      } else {
        if (cmdInput.length() < 30) {
          cmdInput += selected;
        }
      }
      lastButtonPress = millis();
    }
  }
  
  // Плавная анимация курсора
  if (millis() - lastCmdAnim > 16) {
    lastCmdAnim = millis();
    float targetPixelX = 2 + cmdCursorX * 15;
    float targetPixelY = 28 + cmdCursorY * 7;
    cmdSmoothX = cmdSmoothX + (targetPixelX - cmdSmoothX) * 0.2;
    cmdSmoothY = cmdSmoothY + (targetPixelY - cmdSmoothY) * 0.2;
  }
  
  // Отрисовка
  display.clearDisplay();
  if (whiteTheme) {
    display.fillScreen(WHITE);
  } else {
    display.fillScreen(BLACK);
  }
  
  // ========== ЗАГОЛОВОК ==========
  if (whiteTheme) {
    drawRect(0, 0, 128, 10, true);      // чёрный фон
    display.setTextColor(WHITE);         // белый текст
  } else {
    drawRect(0, 0, 128, 10, true);      // белый фон (инвертировано)
    display.setTextColor(BLACK);         // чёрный текст
  }
  display.setCursor(2, 2);
  display.print("SNLDO CMD");
  drawLine(0, 10, 128, 10);
  
  // ========== СТРОКА ВВОДА ==========
  setDrawColor();
  display.setCursor(2, 14);
  display.print("> ");
  display.print(cmdInput);
  if ((millis() / 500) % 2 == 0) display.print("_");
  
  drawLine(0, 22, 128, 22);
  
  // ========== КЛАВИАТУРА ==========
  for (int y = 0; y < 5; y++) {
    for (int x = 0; x < 8; x++) {
      int btnX = 2 + x * 15;
      int btnY = 28 + y * 7;
      setDrawColor();
      display.setCursor(btnX, btnY);
      display.print(cmdKeyboard[y][x]);
    }
  }
  
  // ========== КУРСОР ==========
  int cursorX = (int)cmdSmoothX - 2;
  int cursorY = (int)cmdSmoothY - 1;
  
  if (whiteTheme) {
    display.fillRect(cursorX, cursorY, 14, 8, BLACK);
    display.setTextColor(WHITE);
  } else {
    display.fillRect(cursorX, cursorY, 14, 8, WHITE);
    display.setTextColor(BLACK);
  }
  
  display.setCursor((int)cmdSmoothX, (int)cmdSmoothY);
  display.print(cmdKeyboard[cmdCursorY][cmdCursorX]);
  
  // Возврат цвета
  if (whiteTheme) display.setTextColor(BLACK);
  else display.setTextColor(WHITE);
  
  display.display();
}

//============== SOUND PUD ===============
void playCoinSound() {
  // Звук монетки (восходящий)
  for (int f = 800; f <= 1600; f += 50) {
    tone(SPEAKER_PIN, f, 15);
    delay(10);
  }
  noTone(SPEAKER_PIN);
}

void playBeepSound() {
  // Короткий писк
  tone(SPEAKER_PIN, 1200, 100);
  delay(100);
  noTone(SPEAKER_PIN);
}

void playLaserSound() {
  // Лазер (нисходящий)
  for (int f = 2000; f >= 500; f -= 80) {
    tone(SPEAKER_PIN, f, 10);
    delay(5);
  }
  noTone(SPEAKER_PIN);
}

void playExplosionSound() {
  // Взрыв (шум + падение)
  for (int i = 0; i < 20; i++) {
    tone(SPEAKER_PIN, random(300, 800), 20);
    delay(10);
  }
  for (int f = 500; f >= 100; f -= 30) {
    tone(SPEAKER_PIN, f, 20);
    delay(15);
  }
  noTone(SPEAKER_PIN);
}

void playPowerUpSound() {
  // Повышение уровня
  for (int f = 400; f <= 1200; f += 100) {
    tone(SPEAKER_PIN, f, 50);
    delay(50);
  }
  noTone(SPEAKER_PIN);
}

void playErrorSound() {
  // Ошибка (гудок)
  tone(SPEAKER_PIN, 200, 300);
  delay(300);
  delay(50);
  tone(SPEAKER_PIN, 200, 300);
  delay(300);
  noTone(SPEAKER_PIN);
}

void playBoingSound() {
  // Прыжок (пружина)
  for (int f = 400; f <= 1000; f += 40) {
    tone(SPEAKER_PIN, f, 10);
    delay(5);
  }
  for (int f = 1000; f >= 400; f -= 50) {
    tone(SPEAKER_PIN, f, 10);
    delay(8);
  }
  noTone(SPEAKER_PIN);
}

void playVictorySound() {
  // Победный фанфар
  int melody[] = {523, 659, 784, 1046};
  int durations[] = {200, 200, 200, 400};
  for (int i = 0; i < 4; i++) {
    tone(SPEAKER_PIN, melody[i], durations[i]);
    delay(durations[i] + 30);
  }
  noTone(SPEAKER_PIN);
}

void playSound(int index) {
  if (millis() - lastSoundTime < 300) return; // защита от спама
  
  switch(index) {
    case 0: playCoinSound(); break;
    case 1: playBeepSound(); break;
    case 2: playLaserSound(); break;
    case 3: playExplosionSound(); break;
    case 4: playPowerUpSound(); break;
    case 5: playErrorSound(); break;
    case 6: playBoingSound(); break;
    case 7: playVictorySound(); break;
  }
  lastSoundTime = millis();
}

void runSoundPad() {
  static float soundSmoothY = 0;
  static int soundTargetIndex = 0;
  static unsigned long lastSoundAnim = 0;
  
  if (millis() - lastButtonPress > DEBOUNCE_DELAY) {
    if (digitalRead(BTN_UP) == LOW) {
      if (soundSelection > 0) {
        soundSelection--;
      } else {
        soundSelection = 7;
      }
      soundTargetIndex = soundSelection;
      lastButtonPress = millis();
    }
    
    if (digitalRead(BTN_DOWN) == LOW) {
      if (soundSelection < 7) {
        soundSelection++;
      } else {
        soundSelection = 0;
      }
      soundTargetIndex = soundSelection;
      lastButtonPress = millis();
    }
    
    if (digitalRead(BTN_SELECT) == LOW) {
      playSound(soundSelection);
      lastButtonPress = millis();
    }
    
  }
  
  if (millis() - lastSoundAnim > 16) {
    lastSoundAnim = millis();
    float targetY = 16 + (soundTargetIndex % 4) * 12;
    soundSmoothY = soundSmoothY + (targetY - soundSmoothY) * 0.35;
  }
  
  display.clearDisplay();
  if (whiteTheme) display.fillScreen(WHITE);
  setDrawColor();
  
  drawRect(0, 0, 128, 10, true);
  setInverseColor();
  display.setCursor(2, 2);
  display.print("SOUNDPAD");
  setDrawColor();
  drawLine(0, 10, 128, 10);
  
  int page = soundSelection / 4;
  int start = page * 4;
  display.setCursor(90, 2);
  display.print(page + 1);
  display.print("/2");
  
  for (int i = 0; i < 4; i++) {
    int idx = start + i;
    if (idx < 8) {
      int y = 16 + i * 12;
      setDrawColor();
      display.setCursor(10, y);
      display.print(idx + 1);
      display.print(". ");
      display.print(soundNames[idx]);
    }
  }
  
  int cursorY = (int)soundSmoothY;
  drawRect(5, cursorY - 1, 118, 10, true);
  
  setInverseColor();
  int currentIdx = soundTargetIndex;
  int currentPage = currentIdx / 4;
  int currentRow = currentIdx % 4;
  int currentY = 16 + currentRow * 12;
  
  if (currentPage == page) {
    display.setCursor(10, currentY);
    display.print(currentIdx + 1);
    display.print(". ");
    display.print(soundNames[currentIdx]);
  }
  
  setDrawColor();
  
  display.display();
}

//================== MP3 PLAYER ====================
void runMP3Player() {
  // ========== ИНИЦИАЛИЗАЦИЯ (ОДИН РАЗ) ==========
  if (!mp3Initialized) {
    Serial2.begin(9600, SERIAL_8N1, 16, 17);
    delay(300);
    
    if (myDFPlayer.begin(Serial2)) {
      mp3Initialized = true;
      myDFPlayer.volume(mp3Volume);
      myDFPlayer.EQ(0);
      delay(100);
      
      // ПРИНУДИТЕЛЬНОЕ КОЛИЧЕСТВО ТРЕКОВ
      mp3TotalTracks = MP3_TRACKS;
      
      // ПРОВЕРКА, ЧТО НЕ НОЛЬ
      if (mp3TotalTracks < 1) mp3TotalTracks = 1;
      
      mp3CurrentTrack = 1;
      myDFPlayer.play(mp3CurrentTrack);
      mp3Playing = true;
    } else {
      display.clearDisplay();
      display.setCursor(10, 30);
      display.print("DFPLAYER ERROR!");
      display.display();
      delay(2000);
      changeState(APPS_MENU);
      return;
    }
  }
  
  // ========== АВТОПЕРЕКЛЮЧЕНИЕ ТРЕКОВ ==========
  if (myDFPlayer.available()) {
    if (myDFPlayer.readType() == DFPlayerPlayFinished) {
      // ПРОСТАЯ ЛОГИКА БЕЗ МИНУСОВ
      mp3CurrentTrack = mp3CurrentTrack + 1;
      if (mp3CurrentTrack > mp3TotalTracks) {
        mp3CurrentTrack = 1;
      }
      myDFPlayer.play(mp3CurrentTrack);
      mp3Playing = true;
    }
  }
  
  // ========== УПРАВЛЕНИЕ ==========
  if (millis() - lastButtonPress > DEBOUNCE_DELAY) {
    
    // ГРОМКОСТЬ +
    if (digitalRead(BTN_UP) == LOW && mp3Volume < 30) {
      mp3Volume++;
      myDFPlayer.volume(mp3Volume);
      lastButtonPress = millis();
    }
    
    // ГРОМКОСТЬ -
    if (digitalRead(BTN_DOWN) == LOW && mp3Volume > 0) {
      mp3Volume--;
      myDFPlayer.volume(mp3Volume);
      lastButtonPress = millis();
    }
    
    // ПРЕДЫДУЩИЙ ТРЕК (ВЛЕВО) — ПРОСТАЯ ЛОГИКА
    if (digitalRead(BTN_LEFT) == LOW) {
      mp3CurrentTrack = mp3CurrentTrack - 1;
      if (mp3CurrentTrack < 1) {
        mp3CurrentTrack = mp3TotalTracks;
      }
      myDFPlayer.play(mp3CurrentTrack);
      mp3Playing = true;
      lastButtonPress = millis();
    }
    
    // СЛЕДУЮЩИЙ ТРЕК (ВПРАВО) — ПРОСТАЯ ЛОГИКА
    if (digitalRead(BTN_RIGHT) == LOW) {
      mp3CurrentTrack = mp3CurrentTrack + 1;
      if (mp3CurrentTrack > mp3TotalTracks) {
        mp3CurrentTrack = 1;
      }
      myDFPlayer.play(mp3CurrentTrack);
      mp3Playing = true;
      lastButtonPress = millis();
    }
    
    // PLAY / PAUSE
    if (digitalRead(BTN_SELECT) == LOW) {
      if (mp3Playing) {
        myDFPlayer.pause();
        mp3Playing = false;
      } else {
        myDFPlayer.start();
        mp3Playing = true;
      }
      lastButtonPress = millis();
    }
    
    // ВЫХОД
    if (digitalRead(BTN_BACK) == LOW) {
      myDFPlayer.stop();
      mp3Playing = false;
      changeState(APPS_MENU);
      lastButtonPress = millis();
      return;
    }
  }
  
  // ========== АНИМАЦИЯ ПЛАСТИНКИ ==========
  if (mp3Playing && millis() - lastVinylUpdate > 30) {
    lastVinylUpdate = millis();
    vinylAngle += 6;
    if (vinylAngle >= 360) vinylAngle = 0;
  }
  
  // ========== ОТРИСОВКА ==========
  display.clearDisplay();
  if (whiteTheme) display.fillScreen(WHITE);
  setDrawColor();
  
  drawRect(0, 0, 128, 10, true);
  setInverseColor();
  display.setCursor(2, 2);
  display.print("MP3 PLAYER");
  setDrawColor();
  drawLine(0, 10, 128, 10);
  
  // Пластинка
  drawVinyl(96, 17, vinylAngle);
  
  // Номер трека (ТОЛЬКО ЦИФРА)
  display.setCursor(2, 12);
  display.print("TRACK: ");
  display.print(mp3CurrentTrack);
  
  // Громкость
  display.setCursor(2, 20);
  display.print("VOLUME: ");
  display.print(mp3Volume);
  
  // Статус
  display.setCursor(2, 28 );
  if (mp3Playing) display.print("PLAYING");
  else display.print("PAUSED");
  
  display.setCursor(2, 36);
  display.print("UP/DN:VOL");
  display.setCursor(2, 44);
  display.print("LEFT/RIGHT:TRACK");
 display.display();


}

void drawVinyl(int x, int y, int angle) {
  // Внешний круг (пластинка)
  drawCircle(x + 12, y + 12, 12, false);
  drawCircle(x + 12, y + 12, 10, false);
  
  // Центр (наклейка)
  drawCircle(x + 12, y + 12, 4, true);
  drawCircle(x + 12, y + 12, 2, false);
  
  // Блик (маленький белый кружок)
  drawCircle(x + 8, y + 8, 1, !whiteTheme);
  
  // "Бороздки" (концентрические окружности)
  drawCircle(x + 12, y + 12, 7, false);
  drawCircle(x + 12, y + 12, 5, false);
  
  // Ручка тонарма (линия от центра к краю)
  float rad = radians(angle);
  int lineX = x + 12 + cos(rad) * 10;
  int lineY = y + 12 + sin(rad) * 10;
  drawLine(x + 12, y + 12, lineX, lineY);
}

//=================== 3d render ===================
void projectPoint(float x, float y, float z, float rotX, float rotY, float rotZ, int &screenX, int &screenY) {
  float scale = 10.0;
  float dist = 10.5;
  float centerX = 64, centerY = 32;
  
  // Вращение по X
  float y1 = y * cos(rotX) - z * sin(rotX);
  float z1 = y * sin(rotX) + z * cos(rotX);
  
  // Вращение по Y
  float x1 = x * cos(rotY) + z1 * sin(rotY);
  float z2 = -x * sin(rotY) + z1 * cos(rotY);
  
  // Перспектива
  if (z2 + dist != 0) {
    screenX = centerX + (x1 * scale * dist) / (z2 + dist);
    screenY = centerY + (y1 * scale * dist) / (z2 + dist);
  }
}

void draw3DRender(int shape, float rotX, float rotY) {
  int projected[30][2];
  int vertexCount = 0;
  const int (*edges)[2] = NULL;
  int edgeCount = 0;
  const float (*verts)[3] = NULL;
  
  // Выбор фигуры
  switch(shape) {
    case 0: // Куб
      verts = cubeVerts;
      vertexCount = 8;
      edges = cubeEdges;
      edgeCount = 12;
      break;
    case 1: // Пирамида
      verts = pyramidVerts;
      vertexCount = 5;
      edges = pyramidEdges;
      edgeCount = 8;
      break;
    case 2: // Прямоугольник
      verts = rectVerts;
      vertexCount = 8;
      edges = rectEdges;
      edgeCount = 12;
      break;
  }
  
  // Проецирование
  for (int i = 0; i < vertexCount; i++) {
    projectPoint(verts[i][0], verts[i][1], verts[i][2], rotX, rotY, 0, 
                 projected[i][0], projected[i][1]);
  }
  
  // Рисование рёбер
  for (int i = 0; i < edgeCount; i++) {
    int p1 = edges[i][0];
    int p2 = edges[i][1];
    if (projected[p1][0] > 0 && projected[p1][0] < 128 &&
        projected[p1][1] > 0 && projected[p1][1] < 64 &&
        projected[p2][0] > 0 && projected[p2][0] < 128 &&
        projected[p2][1] > 0 && projected[p2][1] < 64) {
      drawLine(projected[p1][0], projected[p1][1], 
               projected[p2][0], projected[p2][1]);
    }
  }
}

const char* shapeNames[] = {"CUBE", "PYRAMID", "RECT"};

void runRender3D() {
  // ========== FPS ==========
  renderFrameCount++;
  if (millis() - lastRenderFPSTime >= 1000) {
    renderFPS = renderFrameCount;
    renderFrameCount = 0;
    lastRenderFPSTime = millis();
  }
  
  // ========== УПРАВЛЕНИЕ ==========
  if (millis() - lastButtonPress > DEBOUNCE_DELAY) {
    if (digitalRead(BTN_LEFT) == LOW) {
      renderShapeIndex = (renderShapeIndex - 1 + 3) % 3;
      lastButtonPress = millis();
    }
    if (digitalRead(BTN_RIGHT) == LOW) {
      renderShapeIndex = (renderShapeIndex + 1) % 3;
      lastButtonPress = millis();
    }
    if (digitalRead(BTN_UP) == LOW) {
      renderSpeed += 0.5;
      if (renderSpeed > 10.0) renderSpeed = 10.0;
      lastButtonPress = millis();
    }
    if (digitalRead(BTN_DOWN) == LOW) {
      renderSpeed -= 0.5;
      if (renderSpeed < 0.1) renderSpeed = 0.1;
      lastButtonPress = millis();
    }
    if (digitalRead(BTN_BACK) == LOW) {
      changeState(APPS_MENU);
      lastButtonPress = millis();
      return;
    }
  }
  
  // ========== ВРАЩЕНИЕ ==========
  cubeAngleX += 0.02 * renderSpeed;
  cubeAngleY += 0.03 * renderSpeed;
  pyramidAngleX += 0.025 * renderSpeed;
  pyramidAngleY -= 0.02 * renderSpeed;
  rectAngleX += 0.02 * renderSpeed;
  rectAngleY += 0.02 * renderSpeed;
  
  // ========== ОТРИСОВКА ==========
  display.clearDisplay();
  if (whiteTheme) display.fillScreen(WHITE);
  setDrawColor();
  
  // Выбор углов вращения для текущей фигуры
  float rx = 0, ry = 0;
  switch(renderShapeIndex) {
    case 0: rx = cubeAngleX; ry = cubeAngleY; break;
    case 1: rx = pyramidAngleX; ry = pyramidAngleY; break;
    case 2: rx = rectAngleX; ry = rectAngleY; break;
  }
  
  draw3DRender(renderShapeIndex, rx, ry);
  
  // ========== НАЗВАНИЕ ФИГУРЫ ==========
  display.setTextSize(1);
  display.setCursor(50, 2);
  display.print(shapeNames[renderShapeIndex]);
  
  // ========== FPS ==========
  display.setCursor(2, 2);
  display.print("FPS:");
  display.print(renderFPS);
  
  // ========== СКОРОСТЬ ==========
  display.setCursor(2, 47);
  display.print("SPD:");
  display.print(renderSpeed, 1);
  
  // ========== ПОДСКАЗКИ ==========
  display.setCursor(2, 55);
  display.print("L/R:SHAPE UP/DN:SPEED");
  
  display.display();
}

//=============== TETRIS =================
void initTetris() {
  for (int y = 0; y < TETRIS_HEIGHT; y++) {
    for (int x = 0; x < TETRIS_WIDTH; x++) {
      tetrisField[y][x] = 0;
    }
  }
  
  tetrisScore = 0;
  tetrisLines = 0;
  tetrisLevel = 1;
  tetrisSpeed = 1400;
  tetrisActive = true;
  tetrisPaused = false;
  tetrisLastDrop = millis();
  
  // Первая фигура
  currentShapeType = random(0, 7);
  for (int y = 0; y < 4; y++) {
    for (int x = 0; x < 4; x++) {
      currentShape[y][x] = tetrisShapes[currentShapeType][y][x];
    }
  }
  
  // Следующая фигура
  nextShapeType = random(0, 7);
  for (int y = 0; y < 4; y++) {
    for (int x = 0; x < 4; x++) {
      nextShape[y][x] = tetrisShapes[nextShapeType][y][x];
    }
  }
  
  currentX = 3;
  currentY = 0;
}

void newTetrisShape() {
  // Текущая = следующая
  for (int y = 0; y < 4; y++) {
    for (int x = 0; x < 4; x++) {
      currentShape[y][x] = nextShape[y][x];
    }
  }
  currentShapeType = nextShapeType;
  
  // Генерируем следующую
  nextShapeType = random(0, 7);
  for (int y = 0; y < 4; y++) {
    for (int x = 0; x < 4; x++) {
      nextShape[y][x] = tetrisShapes[nextShapeType][y][x];
    }
  }
  
  currentX = 3;
  currentY = 0;
}

bool tetrisCollision(int shape[4][4], int posX, int posY) {
  for (int y = 0; y < 4; y++) {
    for (int x = 0; x < 4; x++) {
      if (shape[y][x] && (posX + x < 0 || posX + x >= TETRIS_WIDTH || 
          posY + y >= TETRIS_HEIGHT || (posY + y >= 0 && tetrisField[posY + y][posX + x]))) {
        return true;
      }
    }
  }
  return false;
}

void lockTetrisShape() {
  for (int y = 0; y < 4; y++) {
    for (int x = 0; x < 4; x++) {
      if (currentShape[y][x]) {
        tetrisField[currentY + y][currentX + x] = 1;
      }
    }
  }
  
  int linesRemoved = 0;
  for (int y = TETRIS_HEIGHT - 1; y >= 0; y--) {
    bool full = true;
    for (int x = 0; x < TETRIS_WIDTH; x++) {
      if (!tetrisField[y][x]) {
        full = false;
        break;
      }
    }
    
    if (full) {
      for (int y2 = y; y2 > 0; y2--) {
        for (int x = 0; x < TETRIS_WIDTH; x++) {
          tetrisField[y2][x] = tetrisField[y2 - 1][x];
        }
      }
      for (int x = 0; x < TETRIS_WIDTH; x++) tetrisField[0][x] = 0;
      linesRemoved++;
      y++;
    }
  }
  
  tetrisLines += linesRemoved;
  tetrisScore += linesRemoved * 1;
  if (tetrisScore > tetrisHighScore) tetrisHighScore = tetrisScore;
  
  if (tetrisLines >= tetrisLevel * 5) {
    tetrisLevel++;
    tetrisSpeed = max(150, 600 - tetrisLevel * 30);
  }
  
  newTetrisShape();
  
  if (tetrisCollision(currentShape, currentX, currentY)) {
    tetrisActive = false;
    saveTetrisHighScore();
  }
}

void rotateTetrisShape() {
  int rotated[4][4] = {0};
  for (int y = 0; y < 4; y++) {
    for (int x = 0; x < 4; x++) {
      rotated[x][3 - y] = currentShape[y][x];
    }
  }
  
  if (!tetrisCollision(rotated, currentX, currentY)) {
    for (int y = 0; y < 4; y++) {
      for (int x = 0; x < 4; x++) {
        currentShape[y][x] = rotated[y][x];
      }
    }
  }
}

void saveTetrisHighScore() {
  if (tetrisHighScore > 255) tetrisHighScore = 255;
  EEPROM.write(EE_TETRIS_HIGH, tetrisHighScore);
  EEPROM.commit();
}

void loadTetrisHighScore() {
  tetrisHighScore = EEPROM.read(EE_TETRIS_HIGH);
}

void drawTetris() {
  // ========== ВЕРХНЯЯ ПАНЕЛЬ ==========
  drawRect(0, 0, 128, 10, true);
  setInverseColor();
  display.setTextSize(1);
  display.setCursor(2, 2);
  display.print("SCORE:");
  display.print(tetrisScore);
  display.setCursor(70, 2);
  display.print("Lv:");
  display.print(tetrisLevel);
  setDrawColor();
  
  // ========== ПОЛЕ ==========
  drawRect(TETRIS_OFFSET_X - 2, TETRIS_OFFSET_Y - 2, TETRIS_WIDTH * 6 + 4, TETRIS_HEIGHT * 6 + 4, false);
  
  for (int y = 0; y < TETRIS_HEIGHT; y++) {
    for (int x = 0; x < TETRIS_WIDTH; x++) {
      int cellX = TETRIS_OFFSET_X + x * 6;
      int cellY = TETRIS_OFFSET_Y + y * 6;
      if (tetrisField[y][x]) {
        display.fillRect(cellX, cellY, 5, 5, whiteTheme ? BLACK : WHITE);
      } else {
        display.drawRect(cellX, cellY, 5, 5, whiteTheme ? BLACK : WHITE);
      }
    }
  }
  
  // ========== ТЕКУЩАЯ ФИГУРА ==========
  for (int y = 0; y < 4; y++) {
    for (int x = 0; x < 4; x++) {
      if (currentShape[y][x]) {
        int cellX = TETRIS_OFFSET_X + (currentX + x) * 6;
        int cellY = TETRIS_OFFSET_Y + (currentY + y) * 6;
        display.fillRect(cellX, cellY, 5, 5, whiteTheme ? BLACK : WHITE);
      }
    }
  }
  
  // ========== СЛЕДУЮЩАЯ ФИГУРА ==========
  display.setCursor(92, 16);
  display.print("NEXT:");
  
  for (int y = 0; y < 4; y++) {
    for (int x = 0; x < 4; x++) {
      if (nextShape[y][x]) {
        int cellX = 96 + x * 6;
        int cellY = 28 + y * 6;
        display.fillRect(cellX, cellY, 5, 5, whiteTheme ? BLACK : WHITE);
      }
    }
  }
  
}

void runTetris() {
  if (!tetrisActive) {
    loadTetrisHighScore();
    initTetris();
    return;
  }
  
 if (gamePaused) {
    drawPauseScreen();
    display.display();
    return;
  }
  
  // ========== УПРАВЛЕНИЕ ==========
  if (millis() - lastButtonPress > DEBOUNCE_DELAY) {
    if (digitalRead(BTN_LEFT) == LOW) {
      if (!tetrisCollision(currentShape, currentX - 1, currentY)) currentX--;
      lastButtonPress = millis();
    }
    if (digitalRead(BTN_RIGHT) == LOW) {
      if (!tetrisCollision(currentShape, currentX + 1, currentY)) currentX++;
      lastButtonPress = millis();
    }
    if (digitalRead(BTN_UP) == LOW) {
      rotateTetrisShape();
      lastButtonPress = millis();
    }
    if (digitalRead(BTN_DOWN) == LOW) {
      if (!tetrisCollision(currentShape, currentX, currentY + 1)) {
        currentY++;
        tetrisLastDrop = millis();
      }
      lastButtonPress = millis();
    }
    if (digitalRead(BTN_SELECT) == LOW) {
      while (!tetrisCollision(currentShape, currentX, currentY + 1)) currentY++;
      lockTetrisShape();
      lastButtonPress = millis();
    }
  }
  
  // ========== ПАДЕНИЕ ==========
  if (millis() - tetrisLastDrop > tetrisSpeed) {
    tetrisLastDrop = millis();
    if (!tetrisCollision(currentShape, currentX, currentY + 1)) {
      currentY++;
    } else {
      lockTetrisShape();
      if (!tetrisActive) {
        saveTetrisHighScore();
        gameOver("TETRIS", tetrisScore, EE_TETRIS_HIGH, tetrisHighScore);
        return;
      }
    }
  }
  
  // ========== ОТРИСОВКА ==========
  display.clearDisplay();
  if (whiteTheme) display.fillScreen(WHITE);
  setDrawColor();
  
  drawTetris();
  
  display.display();

   if (!tetrisActive) {
    if (tetrisScore > tetrisHighScore) {
      tetrisHighScore = tetrisScore;
      EEPROM.write(EE_TETRIS_HIGH, tetrisHighScore);
      EEPROM.commit();
    }
    gameOver("TETRIS", tetrisScore, EE_TETRIS_HIGH, tetrisHighScore);
    return;
  
}
